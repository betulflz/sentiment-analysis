```csharp
using System.Collections.ObjectModel;
using CommunityToolkit.Mvvm.ComponentModel;
using CommunityToolkit.Mvvm.Input;
using SentimentAnalysis.Core;

namespace SentimentAnalysis.Desktop;

/// <summary>Canlı analiz görünümü VM'i. <see cref="IProgress{ChunkResult}"/> olarak pipeline'a verilir;
/// gelen her chunk çizelgeye + güncel VA/duygu/rozete yansır. Start/Stop koşuyu <see cref="IAnalysisRunner"/>
/// üzerinden yönetir. UI-agnostik (headless test).</summary>
public sealed partial class LiveViewModel : ObservableObject, IProgress<ChunkResult>
{
    private readonly IAnalysisRunner _runner;
    private CancellationTokenSource? _cts;
    private Task? _run;

    public ObservableCollection<ChunkRowVm> Timeline { get; } = new();

    [ObservableProperty] private string _currentEmotion = "—";
    [ObservableProperty] private double _currentValence;
    [ObservableProperty] private double _currentArousal;
    [ObservableProperty] private string _congruenceBadge = string.Empty;
    [ObservableProperty] private bool _isCapturing;

    public LiveViewModel(IAnalysisRunner runner) => _runner = runner;

    public void Report(ChunkResult r)
    {
        var row = new ChunkRowVm(r);
        Timeline.Add(row);
        CurrentEmotion = row.Emotion;
        CurrentValence = row.Valence;
        CurrentArousal = row.Arousal;
        CongruenceBadge = row.Badge;
    }

    [RelayCommand]
    private void Start()
    {
        if (IsCapturing) return;
        IsCapturing = true;
        _cts = new CancellationTokenSource();
        _run = RunAsync(_cts.Token);
    }

    private async Task RunAsync(CancellationToken ct)
    {
        try { await _runner.RunAsync(this, ct).ConfigureAwait(true); }
        catch (OperationCanceledException) { /* Stop ile iptal — normal */ }
        finally { IsCapturing = false; }
    }

    [RelayCommand]
    private void Stop() => _cts?.Cancel();

    /// <summary>Test yardımcısı: aktif koşu görevini bekler (yoksa hemen döner).</summary>
    public Task WaitForIdleAsync() => _run ?? Task.CompletedTask;
}
```

> Not: `[ObservableProperty]`/`[RelayCommand]` CommunityToolkit.Mvvm source-generator'ı `CurrentEmotion`, `StartCommand`, `StopCommand` vb. üretir. `ConfigureAwait(true)` — gerçek app'te UI context'e döner; testte senkron context yok, sorun değil (ObservableCollection erişimi test thread'inde). `Report` UI-thread marshalling gerçek app'te `Progress<T>` (yakalanan SynchronizationContext) ile sağlanır — VM `IProgress` implementasyonu doğrudan; app `Report`'u UI thread'ine dispatch eden bir sarmalayıcı geçebilir (08b inceliği; bu tur doğrudan).

/// <summary>Ayarlar VM'i: mevcut <see cref="AppConfig"/>'ten okur, Save ile güncellenmiş config'i callback'e verir
/// (kalıcılık/uygulama composition root'ta). `with { }` ile immutable güncelleme.</summary>
public sealed partial class SettingsViewModel : ObservableObject
{
    private readonly AppConfig _initial;
    private readonly Action<AppConfig> _onSave;

    [ObservableProperty] private int _retentionDays;
    [ObservableProperty] private bool _keepAudio;
    [ObservableProperty] private string _language;

    public SettingsViewModel(AppConfig initial, Action<AppConfig> onSave)
    {
        _initial = initial;
        _onSave = onSave;
        _retentionDays = initial.Storage.AudioRetentionDays;
        _keepAudio = initial.Storage.KeepAudio;
        _language = initial.Stt.Language ?? "tr";
    }

    [RelayCommand]
    private void Save()
    {
        var updated = _initial with
        {
            Storage = _initial.Storage with { AudioRetentionDays = RetentionDays, KeepAudio = KeepAudio },
            Stt = _initial.Stt with { Language = Language },
        };
        _onSave(updated);
    }
}
