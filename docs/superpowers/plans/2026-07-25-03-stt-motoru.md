## Global Constraints

- **Hedef framework:** `net9.0` tüm projeler. Nullable + ImplicitUsings açık.
- **Bağımlılık yönü:** Infrastructure.Stt → Application → Core. Core hiçbir şeye bağlı değil; Infrastructure.Stt yalnızca Application'a (ve dolaylı Core'a) + Whisper.net paketlerine bağlanır.
- **`git YOK`** (proje sahibi kararı): her task'ın son adımı `dotnet build && dotnet test` doğrulamasıdır, `git commit` DEĞİL.
- **STT girdi formatı sabit:** 16 kHz mono float PCM (master D6/spec 03). Motor `SampleRate != 16000` ise `ArgumentException` atar (yeniden örnekleme yakalama katmanının, Alt-proje 05, işidir).
- **`Transcript` domain tipi (Alt-proje 01) DEĞİŞMEZ:** `Transcript(string Text, DetectedLanguage Language, double AvgLogProb)`. Segment-düzeyi zaman EKLENMEZ (chunk-düzeyi zaman `AudioChunk.Start/End`'ten gelir; YAGNI). Spec 03'ün "segment zamanları" ifadesi bu sürümde kapsam dışıdır.
- **`AvgLogProb` semantiği:** Whisper.net `SegmentData.Probability` ortalama token olasılığıdır (0..1). `Transcript.AvgLogProb = log(clamp(ağırlıklı-ort-olasılık, 1e-6, 1))` → negatif; `QualitySignals.TranscriptAvgLogProb` (Alt-proje 02, `LogProbFloor=-1.0`) ile tutarlı.
- **Model varsayılanı:** `SttConfig.LiveModel = "large-v3-turbo"`, `Quantization = "q5_0"` (spec 03 fizibilite ölçümü: M1 8GB'da ~779 MB, RTF 0.17).
- **Gerçek-model entegrasyon testi** `WHISPER_MODEL_PATH` (+ `WHISPER_SAMPLE_WAV`) env değişkenleri yoksa `Assert.Skip` ile atlanır; varsayılan `dotnet test` bunlar olmadan da yeşil kalır.

```csharp
using SentimentAnalysis.Application.Configuration;

namespace SentimentAnalysis.Infrastructure.Stt;

/// <summary>Modeli <c>{ModelDirectory}/ggml-{model}-{quantization}.bin</c> yolundan çözer.
/// ModelDirectory null ise uygulama veri klasörü altındaki varsayılana düşer.</summary>
public sealed class FileSttModelProvider : ISttModelProvider
{
    private readonly SttConfig _cfg;

    public FileSttModelProvider(AppConfig config) => _cfg = config.Stt;

    public string ResolveModelPath(string modelName)
    {
        var dir = _cfg.ModelDirectory ?? DefaultModelDirectory();
        var fileName = $"ggml-{modelName}-{_cfg.Quantization}.bin";
        var path = Path.Combine(dir, fileName);
        if (!File.Exists(path))
            throw new FileNotFoundException(
                $"Whisper modeli bulunamadı: {path}. Modeli indirip bu konuma koyun " +
                $"(huggingface.co/ggerganov/whisper.cpp → {fileName}).", path);
        return path;
    }

    /// <summary>Varsayılan model klasörü: <c>%AppData%/SentimentAnalysis/models</c>
    /// (cross-platform ApplicationData).</summary>
    public static string DefaultModelDirectory() =>
        Path.Combine(
            Environment.GetFolderPath(Environment.SpecialFolder.ApplicationData),
            "SentimentAnalysis", "models");
}
```

```csharp
using SentimentAnalysis.Core;

namespace SentimentAnalysis.Infrastructure.Stt;

/// <summary>Whisper segmentlerini tek bir <see cref="Transcript"/>'e indirger. Saf, deterministik.
/// AvgLogProb = süreyle ağırlıklı ortalama token olasılığının logaritması (negatif; kalite sinyali).</summary>
public static class TranscriptAssembler
{
    /// <summary>Boş transkript (konuşma yok) için avg-logprob tabanı — güçlü "düşük kalite" sinyali.</summary>
    public const double EmptyAvgLogProb = -10.0;

    private const double MinProbability = 1e-6;

    public static Transcript Assemble(IReadOnlyList<SttSegment> segments, DetectedLanguage? forced)
    {
        if (segments.Count == 0)
            return new Transcript(string.Empty, forced ?? DetectedLanguage.Other, EmptyAvgLogProb);

        var text = string.Join(" ", segments
            .Select(s => s.Text.Trim())
            .Where(t => t.Length > 0));

        var language = forced ?? DominantLanguage(segments);
        var avgProb = WeightedAverageProbability(segments);
        var avgLogProb = Math.Log(Math.Clamp(avgProb, MinProbability, 1.0));

        return new Transcript(text, language, avgLogProb);
    }

    /// <summary>En uzun (süreye göre) segmentin dili baskın kabul edilir.</summary>
    private static DetectedLanguage DominantLanguage(IReadOnlyList<SttSegment> segments)
    {
        var code = segments
            .OrderByDescending(s => (s.End - s.Start).TotalSeconds)
            .First().LanguageCode;
        return LanguageMap.FromIso(code);
    }

    /// <summary>Olasılıkları segment süresiyle ağırlıklandırır (sıfır/negatif süre → ağırlık 1).</summary>
    private static double WeightedAverageProbability(IReadOnlyList<SttSegment> segments)
    {
        double weightSum = 0, weighted = 0;
        foreach (var s in segments)
        {
            double w = (s.End - s.Start).TotalSeconds;
            if (w <= 0) w = 1.0;
            weightSum += w;
            weighted += w * Math.Clamp(s.Probability, 0.0, 1.0);
        }
        return weightSum <= 1e-9 ? 0.0 : weighted / weightSum;
    }
}
```

```csharp
using System;
using System.IO;

namespace SentimentAnalysis.Infrastructure.Stt.Tests;

/// <summary>Minimal 16-bit PCM mono WAV okuyucu (yalnızca test için).</summary>
internal static class WavReader
{
    public static float[] ReadMono16(string path)
    {
        using var reader = new BinaryReader(File.OpenRead(path));
        // RIFF başlığını atla, "data" chunk'ını bul.
        reader.ReadBytes(12); // "RIFF"<size>"WAVE"
        while (reader.BaseStream.Position < reader.BaseStream.Length)
        {
            var id = new string(reader.ReadChars(4));
            int size = reader.ReadInt32();
            if (id == "data")
            {
                int count = size / 2;
                var samples = new float[count];
                for (int i = 0; i < count; i++)
                    samples[i] = reader.ReadInt16() / 32768f;
                return samples;
            }
            reader.BaseStream.Seek(size, SeekOrigin.Current);
        }
        throw new InvalidDataException("WAV 'data' chunk bulunamadı.");
    }
}
```

```csharp
/// <summary>Whisper.net (whisper.cpp) tabanlı STT. Model açılışta bir kez yüklenir, sıcak tutulur
/// (singleton). Girdi 16 kHz mono float PCM olmalıdır. İşlemler seri (bounded pipeline tek tüketici,
/// Alt-proje 06) — <see cref="_processLock"/> ile korunur.</summary>
public sealed class WhisperSpeechToText : ISpeechToText, IDisposable
{
    private readonly SttConfig _cfg;
    private readonly ISttModelProvider _models;
    private readonly SemaphoreSlim _initLock = new(1, 1);
    private readonly SemaphoreSlim _processLock = new(1, 1);
    private WhisperFactory? _factory;
    private WhisperProcessor? _processor;
    private bool _disposed;

    public WhisperSpeechToText(AppConfig config, ISttModelProvider models)
    {
        _cfg = config.Stt;
        _models = models;
    }

    public async Task<Transcript> TranscribeAsync(AudioChunk chunk, CancellationToken ct)
    {
        if (chunk.SampleRate != 16000)
            throw new ArgumentException(
                $"Whisper 16 kHz mono bekler; gelen {chunk.SampleRate} Hz. Yeniden örnekleme " +
                "yakalama katmanının (Alt-proje 05) işidir.", nameof(chunk));

        var processor = await EnsureProcessorAsync(ct).ConfigureAwait(false);

        await _processLock.WaitAsync(ct).ConfigureAwait(false);
        try
        {
            var segments = new List<SttSegment>();
            var samples = new ReadOnlyMemory<float>(chunk.Samples);
            await foreach (var seg in processor.ProcessAsync(samples, ct).ConfigureAwait(false))
                segments.Add(new SttSegment(seg.Text, seg.Start, seg.End, seg.Probability, seg.Language));

            DetectedLanguage? forced = _cfg.Language is null ? null : LanguageMap.FromIso(_cfg.Language);
            return TranscriptAssembler.Assemble(segments, forced);
        }
        finally
        {
            _processLock.Release();
        }
    }

    private async Task<WhisperProcessor> EnsureProcessorAsync(CancellationToken ct)
    {
        if (_processor is not null) return _processor;
        await _initLock.WaitAsync(ct).ConfigureAwait(false);
        try
        {
            if (_processor is null)
            {
                var modelPath = _models.ResolveModelPath(_cfg.LiveModel);
                _factory = WhisperFactory.FromPath(modelPath);
                var builder = _factory.CreateBuilder().WithThreads(_cfg.Threads);
                builder = _cfg.Language is null
                    ? builder.WithLanguageDetection()
                    : builder.WithLanguage(_cfg.Language);
                _processor = builder.Build();
            }
            return _processor;
        }
        finally
        {
            _initLock.Release();
        }
    }

    public void Dispose()
    {
        if (_disposed) return;
        _disposed = true;
        _processor?.Dispose();
        _factory?.Dispose();
        _initLock.Dispose();
        _processLock.Dispose();
    }
}
```
